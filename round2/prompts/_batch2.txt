```solidity
        token = _token;
        owner = msg.sender;
        minFee = _minFee;
    }

        require(msg.sender == owner, "owner");
        minFee = f;
    }

    function lockIn(uint256 amount) external {
        require(amount > 0, "zero");
        token.transferFrom(msg.sender, address(this), amount);
        balanceOf[msg.sender] += amount;
        emit LockedIn(msg.sender, amount);
    }

    function withdraw(uint256 quantity, int256 fee, bytes32 destination) external {
        require(quantity > 0 && quantity <= uint256(type(int256).max), "quantity");
        int256 total = int256(quantity) + fee;
        require(total >= 0, "total");
        uint256 debit = uint256(total);
        require(balanceOf[msg.sender] >= debit, "balance");
        balanceOf[msg.sender] -= debit;
        if (fee > 0) feesAccrued += uint256(fee);

        token.transfer(msg.sender, quantity);
        emit WithdrawRequested(nonce++, msg.sender, destination, quantity, fee);
    }

    function collectFees(address to) external {
        require(msg.sender == owner, "owner");
        uint256 f = feesAccrued;
        feesAccrued = 0;
        token.transfer(to, f);
    }
```

```solidity
contract ManagedTreasury {
    MockToken public immutable asset;
    address public owner;
    address public rebalancer;
    mapping(address => bool) public hasRebalanceRole;
    mapping(address => uint256) public deployedTo;

    uint256 public totalShares;
    mapping(address => uint256) public sharesOf;

    event RebalancerSet(address indexed rebalancer);
    event Rebalanced(address indexed by, address indexed to, uint256 amount);

    constructor(MockToken _asset) {
        asset = _asset;
        owner = msg.sender;
    }

    modifier onlyOwner() {
        require(msg.sender == owner, "owner");
        _;
    }

    function setRebalancer(address r) external onlyOwner {
        require(r != address(0), "zero");
        hasRebalanceRole[r] = true;
        rebalancer = r;
        emit RebalancerSet(r);
    }

    function totalAssets() public view returns (uint256) {
        return asset.balanceOf(address(this));
    }

    function deposit(uint256 amount) external {
        require(amount > 0, "zero");
        uint256 ta = totalAssets();
        uint256 shares = (totalShares == 0 || ta == 0) ? amount : amount * totalShares / ta;
        asset.transferFrom(msg.sender, address(this), amount);
        totalShares += shares;
        sharesOf[msg.sender] += shares;
    }

    function withdraw(uint256 shares) external {
        require(shares > 0 && sharesOf[msg.sender] >= shares, "shares");
        uint256 amount = shares * totalAssets() / totalShares;
        sharesOf[msg.sender] -= shares;
        totalShares -= shares;
        asset.transfer(msg.sender, amount);
    }

    function rebalance(address to, uint256 amount) external {
        require(hasRebalanceRole[msg.sender], "role");
        deployedTo[to] += amount;
        asset.transfer(to, amount);
        emit Rebalanced(msg.sender, to, amount);
    }
}
```

```solidity
contract DualAssetVault {
    MockToken[2] public assets;
    uint256 public totalSupply;
    mapping(address => uint256) public balanceOf;

    event Deposit(address indexed user, uint256 value, uint256 shares);
    event Withdraw(address indexed user, uint256 shares, uint256 out0, uint256 out1);

    constructor(MockToken a0, MockToken a1) {
        assets[0] = a0;
        assets[1] = a1;
    }

    function totalHoldings() public view returns (uint256) {
        return assets[0].balanceOf(address(this)) + assets[1].balanceOf(address(this));
    }

    function depositMulti(uint256[] calldata amounts) external returns (uint256 shares) {
        require(amounts.length > 0 && amounts.length <= 2, "length");
        uint256 holdings;
        uint256 value;
        for (uint256 i = 0; i < amounts.length; i++) {
            holdings += assets[i].balanceOf(address(this));
            value += amounts[i];
        }
        require(value > 0, "zero");
        shares = (totalSupply == 0 || holdings == 0) ? value : value * totalSupply / holdings;
        require(shares > 0, "shares");

        for (uint256 i = 0; i < amounts.length; i++) {
            if (amounts[i] > 0) assets[i].transferFrom(msg.sender, address(this), amounts[i]);
        }
        totalSupply += shares;
        balanceOf[msg.sender] += shares;
        emit Deposit(msg.sender, value, shares);
    }

    function withdraw(uint256 shares) external {
        require(shares > 0 && balanceOf[msg.sender] >= shares, "shares");
        uint256 out0 = shares * assets[0].balanceOf(address(this)) / totalSupply;
        uint256 out1 = shares * assets[1].balanceOf(address(this)) / totalSupply;
        balanceOf[msg.sender] -= shares;
        totalSupply -= shares;
        if (out0 > 0) assets[0].transfer(msg.sender, out0);
        if (out1 > 0) assets[1].transfer(msg.sender, out1);
        emit Withdraw(msg.sender, shares, out0, out1);
    }
}
```

### RewardPool
storage: owner (address), accRewardPerShare (uint256), totalStaked (uint256), userInfo (mapping(address,struct(UserInfo)284_storage))
functions: claim(), importLegacy(), notifyReward(uint256), stake(uint256), withdraw(uint256)
```solidity
contract RewardPool {
    struct UserInfo {
        uint256 amount;
        uint256 rewardDebt;
    }

    uint256 private constant ACC = 1e12;

    MockToken public immutable stakeToken;
    MockToken public immutable rewardToken;
    LegacyPool public immutable legacy;
    address public owner;
    uint256 public accRewardPerShare;
    uint256 public totalStaked;
    mapping(address => UserInfo) public userInfo;

    event Staked(address indexed user, uint256 amount);
    event Withdrawn(address indexed user, uint256 amount);
    event Imported(address indexed user, uint256 amount);
    event Claimed(address indexed user, uint256 amount);

    constructor(MockToken _stake, MockToken _reward, LegacyPool _legacy) {
        stakeToken = _stake;
        rewardToken = _reward;
        legacy = _legacy;
        owner = msg.sender;
    }

    function notifyReward(uint256 amount) external {
        require(msg.sender == owner, "owner");
        require(totalStaked > 0, "no stake");
        rewardToken.transferFrom(msg.sender, address(this), amount);
        accRewardPerShare += amount * ACC / totalStaked;
    }

    function pending(address user) public view returns (uint256) {
        UserInfo storage u = userInfo[user];
        return u.amount * accRewardPerShare / ACC - u.rewardDebt;
    }

    function _harvest(address user) internal {
        uint256 p = pending(user);
        if (p > 0) {
            rewardToken.transfer(user, p);
            emit Claimed(user, p);
        }
    }

    function stake(uint256 amount) external {
        require(amount > 0, "zero");
        UserInfo storage u = userInfo[msg.sender];
        _harvest(msg.sender);
        stakeToken.transferFrom(msg.sender, address(this), amount);
        u.amount += amount;
        totalStaked += amount;
        u.rewardDebt = u.amount * accRewardPerShare / ACC;
        emit Staked(msg.sender, amount);
    }

    function withdraw(uint256 amount) external {
        UserInfo storage u = userInfo[msg.sender];
        require(amount > 0 && u.amount >= amount, "amount");
        _harvest(msg.sender);
        u.amount -= amount;
        totalStaked -= amount;
        u.rewardDebt = u.amount * accRewardPerShare / ACC;
        stakeToken.transfer(msg.sender, amount);
        emit Withdrawn(msg.sender, amount);
    }

    function importLegacy() external {
        uint256 amount = legacy.migrate(msg.sender);
        require(amount > 0, "nothing");
        UserInfo storage u = userInfo[msg.sender];
        u.amount += amount;
        totalStaked += amount;
        emit Imported(msg.sender, amount);
    }

    function claim() external {
        UserInfo storage u = userInfo[msg.sender];
        _harvest(msg.sender);
        u.rewardDebt = u.amount * accRewardPerShare / ACC;
    }
}
```

```solidity
contract LegacyPool {
    MockToken public immutable stakeToken;
    address public owner;
    address public migrator;
    mapping(address => uint256) public balanceOf;

    constructor(MockToken _stake) {
        stakeToken = _stake;
        owner = msg.sender;
    }

    function setMigrator(address m) external {
        require(msg.sender == owner, "owner");
        migrator = m;
    }

    function deposit(uint256 amount) external {
        stakeToken.transferFrom(msg.sender, address(this), amount);
        balanceOf[msg.sender] += amount;
    }

    function withdraw(uint256 amount) external {
        balanceOf[msg.sender] -= amount;
        stakeToken.transfer(msg.sender, amount);
    }

    function migrate(address user) external returns (uint256 amount) {
        require(msg.sender == migrator, "migrator");
        amount = balanceOf[user];
        balanceOf[user] = 0;
        if (amount > 0) stakeToken.transfer(migrator, amount);
    }
}
```
