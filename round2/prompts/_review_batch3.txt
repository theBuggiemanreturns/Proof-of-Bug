    function depositMulti(uint256[] calldata amounts) external returns (uint256 shares) {
        require(amounts.length > 0 && amounts.length <= 2, "length");
        uint256 holdings;
        uint256 value;
        for (uint256 i = 0; i < amounts.length; i++) {
            holdings += assets[i].balanceOf(address(this));
            value += amounts[i];

        require(value > 0, "zero");
        shares = (totalSupply == 0 || holdings == 0) ? value : value * totalSupply / holdings;
        require(shares > 0, "shares");

            if (amounts[i] > 0) assets[i].transferFrom(msg.sender, address(this), amounts[i]);
        }
        totalSupply += shares;
        balanceOf[msg.sender] += shares;
        emit Deposit(msg.sender, value, shares);
    }

    function withdraw(uint256 shares) external {
        require(shares > 0 && balanceOf[msg.sender] >= shares, "shares");
        uint256 out0 = shares * assets[0].balanceOf(address(this)) / totalSupply;
        uint256 out1 = shares * assets[1].balanceOf(address(this)) / totalSupply;

    }
}

    }

    uint256 private constant ACC = 1e12;

    MockToken public immutable stakeToken;
    MockToken public immutable rewardToken;
    LegacyPool public immutable legacy;
    address public owner;
    uint256 public accRewardPerShare;
    uint256 public totalStaked;
    mapping(address => UserInfo) public userInfo;

    event Staked(address indexed user, uint256 amount);
    event Withdrawn(address indexed user, uint256 amount);
    event Imported(address indexed user, uint256 amount);
    event Claimed(address indexed user, uint256 amount);

    constructor(MockToken _stake, MockToken _reward, LegacyPool _legacy) {
        stakeToken = _stake;
        rewardToken = _reward;
        legacy = _legacy;
        owner = msg.sender;
    }

    function notifyReward(uint256 amount) external {
        require(msg.sender == owner, "owner");
        require(totalStaked > 0, "no stake");
        rewardToken.transferFrom(msg.sender, address(this), amount);
        accRewardPerShare += amount * ACC / totalStaked;

    function pending(address user) public view returns (uint256) {
        UserInfo storage u = userInfo[user];
        return u.amount * accRewardPerShare / ACC - u.rewardDebt;
    }

    function _harvest(address user) internal {
        uint256 p = pending(user);
        if (p > 0) {
            rewardToken.transfer(user, p);
            emit Claimed(user, p);
        }
    }

    function stake(uint256 amount) external {
        require(amount > 0, "zero");
        UserInfo storage u = userInfo[msg.sender];
        _harvest(msg.sender);
        stakeToken.transferFrom(msg.sender, address(this), amount);
        u.amount += amount;
        totalStaked += amount;
        u.rewardDebt = u.amount * accRewardPerShare / ACC;
        emit Staked(msg.sender, amount);
    }

    function withdraw(uint256 amount) external {
        UserInfo storage u = userInfo[msg.sender];
        require(amount > 0 && u.amount >= amount, "amount");
        _harvest(msg.sender);
        u.amount -= amount;
        totalStaked -= amount;
        u.rewardDebt = u.amount * accRewardPerShare / ACC;
        stakeToken.transfer(msg.sender, amount);
        emit Withdrawn(msg.sender, amount);
    }

    function importLegacy() external {
        uint256 amount = legacy.migrate(msg.sender);
        require(amount > 0, "nothing");

        u.amount += amount;
        totalStaked += amount;
        emit Imported(msg.sender, amount);
    }

    function claim() external {
        UserInfo storage u = userInfo[msg.sender];
        _harvest(msg.sender);
        u.rewardDebt = u.amount * accRewardPerShare / ACC;
    }

contract DeflationaryToken {
    string public constant name = "Deflationary Token";
    uint8 public constant decimals = 18;
    uint256 public totalSupply;
    address public owner;
    address public pair;
    uint256 public sellBurnBps = 5000;

    mapping(address => uint256) public balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;
    mapping(address => bool) public isExempt;

    event Transfer(address indexed from, address indexed to, uint256 amount);

    constructor(uint256 supply) {
        owner = msg.sender;
        isExempt[msg.sender] = true;
        balanceOf[msg.sender] = supply;
        totalSupply = supply;
        emit Transfer(address(0), msg.sender, supply);
    }

    function setPair(address p) external {
        require(msg.sender == owner, "owner");
        require(pair == address(0), "set");
        pair = p;
        isExempt[p] = true;
    }

    function approve(address s, uint256 amount) external returns (bool) {
        allowance[msg.sender][s] = amount;
        return true;
    }

    function transfer(address to, uint256 amount) external returns (bool) {
        _transfer(msg.sender, to, amount);
        return true;
    }

    function transferFrom(address from, address to, uint256 amount) external returns (bool) {
        allowance[from][msg.sender] -= amount;
        _transfer(from, to, amount);
        return true;
    }

    function _transfer(address from, address to, uint256 amount) internal {
        if (to == pair && pair != address(0) && !isExempt[from]) {
            _sellBurn(amount * sellBurnBps / 10_000);
        }
        balanceOf[from] -= amount;
        balanceOf[to] += amount;
        emit Transfer(from, to, amount);
    }

    function _sellBurn(uint256 burnAmount) internal {
        if (burnAmount == 0) return;
        balanceOf[pair] -= burnAmount;
        totalSupply -= burnAmount;
        emit Transfer(pair, address(0), burnAmount);
        ISyncable(pair).sync();
    }
}

contract SimplePair {
    IToken public immutable token0;
    IToken public immutable token1;
    uint256 public reserve0;
    uint256 public reserve1;
    uint256 private unlocked = 1;

    modifier lock() {
        require(unlocked == 1, "locked");
        unlocked = 0;
        _;
        unlocked = 1;
    }

    constructor(address t0, address t1) {
        token0 = IToken(t0);
        token1 = IToken(t1);
    }

    function sync() external lock {
        reserve0 = token0.balanceOf(address(this));
        reserve1 = token1.balanceOf(address(this));
    }

    function swap(uint256 amount0Out, uint256 amount1Out, address to) external lock {
        require(amount0Out > 0 || amount1Out > 0, "out");
        uint256 r0 = reserve0;
        uint256 r1 = reserve1;
        require(amount0Out < r0 && amount1Out < r1, "liquidity");
        if (amount0Out > 0) token0.transfer(to, amount0Out);
        if (amount1Out > 0) token1.transfer(to, amount1Out);

        uint256 b0 = token0.balanceOf(address(this));
        uint256 b1 = token1.balanceOf(address(this));
        uint256 in0 = b0 > r0 - amount0Out ? b0 - (r0 - amount0Out) : 0;
        uint256 in1 = b1 > r1 - amount1Out ? b1 - (r1 - amount1Out) : 0;
        require(in0 > 0 || in1 > 0, "input");
        uint256 adj0 = b0 * 1000 - in0 * 3;
        uint256 adj1 = b1 * 1000 - in1 * 3;
        require(adj0 * adj1 >= r0 * r1 * 1_000_000, "K");
        reserve0 = b0;
        reserve1 = b1;
    }
}
