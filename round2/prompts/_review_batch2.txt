```solidity
contract StakeLedger {
    address public admin;
    uint256 public totalSupply;
    uint256 public totalDelegated;

    mapping(address => uint256) public balanceOf;   // spendable
    mapping(address => uint256) public locked;      // vesting, not spendable
    mapping(address => uint256) public delegated;

    event Transfer(address indexed from, address indexed to, uint256 amount);
    event Delegated(address indexed account, uint256 amount);
    event Undelegated(address indexed account, uint256 amount);

    constructor() {
        admin = msg.sender;
    }

    modifier onlyAdmin() {
        require(msg.sender == admin, "admin");
        _;
    }

    function credit(address to, uint256 amount) external onlyAdmin {
        balanceOf[to] += amount;
        totalSupply += amount;
        emit Transfer(address(0), to, amount);
    }

    function grantVesting(address to, uint256 amount) external onlyAdmin {
        locked[to] += amount;
        totalSupply += amount;
    }

    function delegate(uint256 amount) external {
        require(amount > 0, "zero");
        require(amount <= balanceOf[msg.sender] + locked[msg.sender] - delegated[msg.sender], "insufficient");
        delegated[msg.sender] += amount;
        totalDelegated += amount;
        unchecked {
            balanceOf[msg.sender] -= amount;
        }
        emit Delegated(msg.sender, amount);
    }

    function undelegate(uint256 amount) external {
        require(amount > 0 && amount <= delegated[msg.sender], "amount");
        delegated[msg.sender] -= amount;
        totalDelegated -= amount;
        unchecked {
            balanceOf[msg.sender] += amount;
        }
        emit Undelegated(msg.sender, amount);
    }

    function transfer(address to, uint256 amount) external returns (bool) {
        require(to != address(0), "to");
        require(balanceOf[msg.sender] >= amount, "balance");
        unchecked {
            balanceOf[msg.sender] -= amount;
            balanceOf[to] += amount;
        }
        emit Transfer(msg.sender, to, amount);
        return true;
    }
}
```

```solidity
contract BridgeGateway {
    MockToken public immutable token;
    address public owner;
    int256 public minFee;
    uint256 public feesAccrued;
    uint256 public nonce;
    mapping(address => uint256) public balanceOf;

    event LockedIn(address indexed user, uint256 amount);
    event WithdrawRequested(uint256 indexed nonce, address indexed user, bytes32 destination, uint256 quantity, int256 fee);

    constructor(MockToken _token, int256 _minFee) {
        token = _token;
        owner = msg.sender;
        minFee = _minFee;
    }

    function setMinFee(int256 f) external {
        require(msg.sender == owner, "owner");
        minFee = f;
    }

    function lockIn(uint256 amount) external {
        require(amount > 0, "zero");
        token.transferFrom(msg.sender, address(this), amount);
        balanceOf[msg.sender] += amount;
        emit LockedIn(msg.sender, amount);
    }

    function withdraw(uint256 quantity, int256 fee, bytes32 destination) external {
        require(quantity > 0 && quantity <= uint256(type(int256).max), "quantity");
        int256 total = int256(quantity) + fee;
        require(total >= 0, "total");
        uint256 debit = uint256(total);
        require(balanceOf[msg.sender] >= debit, "balance");
        balanceOf[msg.sender] -= debit;
        if (fee > 0) feesAccrued += uint256(fee);

        token.transfer(msg.sender, quantity);
        emit WithdrawRequested(nonce++, msg.sender, destination, quantity, fee);
    }

    function collectFees(address to) external {
        require(msg.sender == owner, "owner");
        uint256 f = feesAccrued;
        feesAccrued = 0;
        token.transfer(to, f);
    }
}
```

### ManagedTreasury
storage: owner (address), rebalancer (address), hasRebalanceRole (mapping(address,bool)), deployedTo (mapping(address,uint256)), totalShares (uint256), sharesOf (mapping(address,uint256))
functions: deposit(uint256), rebalance(address,uint256), setRebalancer(address), withdraw(uint256)
```solidity
contract ManagedTreasury {
    MockToken public immutable asset;
    address public owner;
    address public rebalancer;
    mapping(address => bool) public hasRebalanceRole;
    mapping(address => uint256) public deployedTo;

    uint256 public totalShares;
    mapping(address => uint256) public sharesOf;

    event RebalancerSet(address indexed rebalancer);
    event Rebalanced(address indexed by, address indexed to, uint256 amount);

    constructor(MockToken _asset) {
        asset = _asset;
        owner = msg.sender;
    }

    modifier onlyOwner() {
        require(msg.sender == owner, "owner");
        _;
    }

    function setRebalancer(address r) external onlyOwner {
        require(r != address(0), "zero");
        hasRebalanceRole[r] = true;
        rebalancer = r;
        emit RebalancerSet(r);
    }

    function totalAssets() public view returns (uint256) {
        return asset.balanceOf(address(this));
    }

    function deposit(uint256 amount) external {
        require(amount > 0, "zero");
        uint256 ta = totalAssets();
        uint256 shares = (totalShares == 0 || ta == 0) ? amount : amount * totalShares / ta;
        asset.transferFrom(msg.sender, address(this), amount);
        totalShares += shares;
        sharesOf[msg.sender] += shares;
    }

    function withdraw(uint256 shares) external {
        require(shares > 0 && sharesOf[msg.sender] >= shares, "shares");
        uint256 amount = shares * totalAssets() / totalShares;
        sharesOf[msg.sender] -= shares;
        totalShares -= shares;
        asset.transfer(msg.sender, amount);
    }

    function rebalance(address to, uint256 amount) external {
        require(hasRebalanceRole[msg.sender], "role");
        deployedTo[to] += amount;
        asset.transfer(to, amount);
        emit Rebalanced(msg.sender, to, amount);
    }
}
```
