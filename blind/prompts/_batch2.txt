        return asset.balanceOf(address(this));

        return bonusOwed[account] + (balanceOf[account] * accBonusPerShare) / ACC - bonusDebt[account];
    }

        if (account == address(0)) return;
        bonusOwed[account] = pendingBonus(account);
    }

        if (account == address(0)) return;
        bonusDebt[account] = (balanceOf[account] * accBonusPerShare) / ACC;

        require(shares > 0, "zero shares");
        require(asset.transferFrom(msg.sender, address(this), assets), "transfer");

        totalSupply = supply + shares;
        balanceOf[msg.sender] += shares;
        _sync(msg.sender);
        emit Transfer(address(0), msg.sender, shares);
        emit Deposit(msg.sender, assets, shares);
    }

        require(shares > 0, "zero");
        require(balanceOf[msg.sender] >= shares, "balance");
        assets = (shares * totalAssets()) / totalSupply;
        _accrue(msg.sender);
        balanceOf[msg.sender] -= shares;

contract OrderBookC07 {
    struct Order {
        address maker;
        uint128 amount;
        uint128 price;
        bool active;
    }

    uint256 public constant PRICE_SCALE = 1e18;

    TokenC07 public immutable base;
    TokenC07 public immutable quote;

    uint256 public nextOrderId = 1;
    mapping(uint256 => Order) public orders;

    event Placed(uint256 indexed id, address indexed maker, uint256 amount, uint256 price);
    event Filled(uint256 indexed id, address indexed taker, uint256 baseAmount, uint256 quoteAmount);
    event Cancelled(uint256 indexed id, uint256 refunded);
    event Modified(uint256 indexed id, uint256 newAmount, uint256 newPrice);

    constructor(TokenC07 _base, TokenC07 _quote) {
        base = _base;
        quote = _quote;
    }

    function place(uint128 amount, uint128 price) external returns (uint256 id) {
        require(amount > 0 && price > 0, "params");
        require(base.transferFrom(msg.sender, address(this), amount), "transfer");
        id = nextOrderId++;
        orders[id] = Order({maker: msg.sender, amount: amount, price: price, active: true});
        emit Placed(id, msg.sender, amount, price);
    }

    function quoteFor(uint256 id, uint256 baseAmount) public view returns (uint256) {
        uint256 cost = baseAmount * orders[id].price;
        return (cost + PRICE_SCALE - 1) / PRICE_SCALE;
    }

    function fill(uint256 id, uint128 baseAmount, uint256 maxQuote) external returns (uint256 quoteAmount) {
        Order storage o = orders[id];
        require(o.active, "inactive");
        require(baseAmount > 0 && baseAmount <= o.amount, "amount");
        quoteAmount = quoteFor(id, baseAmount);
        require(quoteAmount <= maxQuote, "slippage");
        o.amount -= baseAmount;
        if (o.amount == 0) {
            o.active = false;
        }
        require(quote.transferFrom(msg.sender, o.maker, quoteAmount), "quote transfer");
        require(base.transfer(msg.sender, baseAmount), "base transfer");
        emit Filled(id, msg.sender, baseAmount, quoteAmount);
    }

    function cancel(uint256 id) external {
        Order storage o = orders[id];
        require(o.maker == msg.sender, "maker");
        require(o.active, "inactive");
        o.active = false;
        uint256 refund = o.amount;
        require(base.transfer(msg.sender, refund), "transfer");
        emit Cancelled(id, refund);
    }

    function modify(uint256 id, uint128 newAmount, uint128 newPrice) external {
        Order storage o = orders[id];
        require(o.maker == msg.sender, "maker");
        require(newAmount > 0 && newPrice > 0, "params");
        uint128 current = o.amount;
        o.amount = newAmount;
        o.price = newPrice;
        if (newAmount > current) {
            require(base.transferFrom(msg.sender, address(this), newAmount - current), "transfer");
        } else if (newAmount < current) {
            require(base.transfer(msg.sender, current - newAmount), "transfer");
        }
        emit Modified(id, newAmount, newPrice);
    }
}

    event Sync(uint112 reserve0, uint112 reserve1);
    event Transfer(address indexed from, address indexed to, uint256 value);

    modifier lock() {
        require(unlocked == 1, "LOCKED");
        unlocked = 0;
        _;
        unlocked = 1;
    }

    constructor(TokenC08 _token0, TokenC08 _token1) {
        token0 = _token0;
        token1 = _token1;
    }

    function getReserves() public view returns (uint112 _reserve0, uint112 _reserve1, uint32 _blockTimestampLast) {
        _reserve0 = reserve0;
        _reserve1 = reserve1;
        _blockTimestampLast = blockTimestampLast;
    }

    function _update(uint256 balance0, uint256 balance1) private {
        require(balance0 <= type(uint112).max && balance1 <= type(uint112).max, "OVERFLOW");
        reserve0 = uint112(balance0);
        reserve1 = uint112(balance1);
        blockTimestampLast = uint32(block.timestamp);
        emit Sync(reserve0, reserve1);
    }

    function _mint(address to, uint256 value) private {
        totalSupply += value;
        balanceOf[to] += value;
        emit Transfer(address(0), to, value);
    }

    function _burn(address from, uint256 value) private {
        balanceOf[from] -= value;
        totalSupply -= value;
        emit Transfer(from, address(0), value);
    }

    function transfer(address to, uint256 value) external returns (bool) {
        balanceOf[msg.sender] -= value;
        balanceOf[to] += value;
        emit Transfer(msg.sender, to, value);
        return true;
    }

    function mint(address to) external lock returns (uint256 liquidity) {
        (uint112 _reserve0, uint112 _reserve1, ) = getReserves();
        uint256 balance0 = token0.balanceOf(address(this));
        uint256 balance1 = token1.balanceOf(address(this));
        uint256 amount0 = balance0 - _reserve0;
        uint256 amount1 = balance1 - _reserve1;

        uint256 _totalSupply = totalSupply;
        if (_totalSupply == 0) {
            liquidity = _sqrt(amount0 * amount1) - MINIMUM_LIQUIDITY;
            _mint(address(0xdead), MINIMUM_LIQUIDITY);
        } else {
            uint256 l0 = (amount0 * _totalSupply) / _reserve0;
            uint256 l1 = (amount1 * _totalSupply) / _reserve1;
            liquidity = l0 < l1 ? l0 : l1;
        }
        require(liquidity > 0, "INSUFFICIENT_LIQUIDITY_MINTED");
        _mint(to, liquidity);
        _update(balance0, balance1);
        emit Mint(msg.sender, amount0, amount1);

        uint256 balance0 = token0.balanceOf(address(this));
        uint256 balance1 = token1.balanceOf(address(this));
        uint256 liquidity = balanceOf[address(this)];
        uint256 _totalSupply = totalSupply;
        amount0 = (liquidity * balance0) / _totalSupply;
        amount1 = (liquidity * balance1) / _totalSupply;

        _burn(address(this), liquidity);
        require(token0.transfer(to, amount0), "T0");
        require(token1.transfer(to, amount1), "T1");
        _update(token0.balanceOf(address(this)), token1.balanceOf(address(this)));
        emit Burn(msg.sender, amount0, amount1, to);

        require(amount0Out > 0 || amount1Out > 0, "INSUFFICIENT_OUTPUT_AMOUNT");
        (uint112 _reserve0, uint112 _reserve1, ) = getReserves();
        require(amount0Out < _reserve0 && amount1Out < _reserve1, "INSUFFICIENT_LIQUIDITY");

        if (amount0Out > 0) require(token0.transfer(to, amount0Out), "T0");
        if (amount1Out > 0) require(token1.transfer(to, amount1Out), "T1");

        uint256 balance0 = token0.balanceOf(address(this));
        uint256 balance1 = token1.balanceOf(address(this));
        uint256 amount0In = balance0 > _reserve0 - amount0Out ? balance0 - (_reserve0 - amount0Out) : 0;
        uint256 amount1In = balance1 > _reserve1 - amount1Out ? balance1 - (_reserve1 - amount1Out) : 0;
        require(amount0In > 0 || amount1In > 0, "INSUFFICIENT_INPUT_AMOUNT");

        uint256 balance0Adjusted = balance0 * 10_000 - amount0In * FEE_BPS;
        uint256 balance1Adjusted = balance1 * 10_000 - amount1In * FEE_BPS;
        require(balance0Adjusted * balance1Adjusted >= uint256(_reserve0) * _reserve1 * 1000 ** 2, "K");

        _update(balance0, balance1);
        emit Swap(msg.sender, amount0In, amount1In, amount0Out, amount1Out, to);
    }

    function skim(address to) external lock {
        require(token0.transfer(to, token0.balanceOf(address(this)) - reserve0), "T0");
        require(token1.transfer(to, token1.balanceOf(address(this)) - reserve1), "T1");
    }

    function sync() external lock {
        _update(token0.balanceOf(address(this)), token1.balanceOf(address(this)));
    }

    function _sqrt(uint256 y) internal pure returns (uint256 z) {
        if (y > 3) {
            z = y;
            uint256 x = y / 2 + 1;
            while (x < z) {
                z = x;
                x = (y / x + x) / 2;
            }
        } else if (y != 0) {
            z = 1;
        }
    }
}

    }

    function previewMint(uint256 shares) public view returns (uint256) {
        return _toAssets(shares, true);
    }

    function previewRedeem(uint256 shares) public view returns (uint256) {
        uint256 gross = _toAssets(shares, false);
        return gross - _mulDiv(gross, EXIT_FEE_BPS, BPS, true);
    }

    function previewWithdraw(uint256 assets) public view returns (uint256) {
        uint256 gross = _mulDiv(assets, BPS, BPS - EXIT_FEE_BPS, true);
        return _toShares(gross, true);
    }

    function deposit(uint256 assets, address receiver) external returns (uint256 shares) {
        shares = previewDeposit(assets);
        require(shares > 0, "zero shares");
        _enter(receiver, assets, shares);
    }

    function mint(uint256 shares, address receiver) external returns (uint256 assets) {
        require(shares > 0, "zero shares");
        assets = previewMint(shares);
        _enter(receiver, assets, shares);
    }

    function redeem(uint256 shares, address receiver, address owner) external returns (uint256 assets) {
        assets = previewRedeem(shares);
        require(assets > 0, "zero assets");
        _exit(owner, receiver, assets, shares);
    }

    function withdraw(uint256 assets, address receiver, address owner) external returns (uint256 shares) {
        require(assets > 0, "zero assets");
        shares = previewWithdraw(assets);
        _exit(owner, receiver, assets, shares);
    }

    function _enter(address receiver, uint256 assets, uint256 shares) internal {
        require(asset.transferFrom(msg.sender, address(this), assets), "transfer");
        totalSupply += shares;
        balanceOf[receiver] += shares;
        emit Transfer(address(0), receiver, shares);
        emit Deposit(msg.sender, receiver, assets, shares);
    }

    function _exit(address owner, address receiver, uint256 assets, uint256 shares) internal {
        if (msg.sender != owner) {
            uint256 allowed = allowance[owner][msg.sender];
            if (allowed != type(uint256).max) {
                require(allowed >= shares, "allowance");
                allowance[owner][msg.sender] = allowed - shares;
            }
        }
        uint256 gross = _toAssets(shares, false);
        uint256 fee = gross > assets ? gross - assets : 0;
        require(balanceOf[owner] >= shares, "balance");
        balanceOf[owner] -= shares;
        totalSupply -= shares;
        emit Transfer(owner, address(0), shares);
        emit Withdraw(msg.sender, receiver, owner, assets, shares);
        require(asset.transfer(receiver, assets), "transfer");
        if (fee > 0) {
            require(asset.transfer(feeRecipient, fee), "fee");
        }
    }

    function approve(address spender, uint256 amount) external returns (bool) {
        allowance[msg.sender][spender] = amount;
        emit Approval(msg.sender, spender, amount);
        return true;
    }

    function transfer(address to, uint256 amount) external returns (bool) {
        require(balanceOf[msg.sender] >= amount, "balance");
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        emit Transfer(msg.sender, to, amount);
        return true;
    }
}
