contract FarmC10 {
    struct UserInfo {
        uint256 amount;
        uint256 rewardDebt;
    }

    uint256 private constant ACC_PRECISION = 1e12;

    TokenC10 public immutable lpToken;
    TokenC10 public immutable rewardToken;
    address public owner;

    uint256 public rewardPerSecond;
    uint256 public lastRewardTime;
    uint256 public accRewardPerShare;
    uint256 public totalStaked;

    mapping(address => UserInfo) public userInfo;

    event Deposit(address indexed user, uint256 amount);
    event Withdraw(address indexed user, uint256 amount);
    event Harvest(address indexed user, uint256 amount);
    event EmergencyWithdraw(address indexed user, uint256 amount);
    event RewardRateUpdated(uint256 rewardPerSecond);

    constructor(TokenC10 _lpToken, TokenC10 _rewardToken, uint256 _rewardPerSecond) {
        lpToken = _lpToken;
        rewardToken = _rewardToken;
        rewardPerSecond = _rewardPerSecond;
        lastRewardTime = block.timestamp;
        owner = msg.sender;
    }

    function pendingReward(address account) external view returns (uint256) {
        UserInfo storage user = userInfo[account];
        uint256 acc = accRewardPerShare;
        if (block.timestamp > lastRewardTime && totalStaked != 0) {
            acc += ((block.timestamp - lastRewardTime) * rewardPerSecond * ACC_PRECISION) / totalStaked;
        }
        return (user.amount * acc) / ACC_PRECISION - user.rewardDebt;
    }

    function updatePool() public {
        if (block.timestamp <= lastRewardTime) return;
        if (totalStaked == 0) {
            lastRewardTime = block.timestamp;
            return;
        }
        uint256 reward = (block.timestamp - lastRewardTime) * rewardPerSecond;
        accRewardPerShare += (reward * ACC_PRECISION) / totalStaked;
        lastRewardTime = block.timestamp;
    }

    function deposit(uint256 amount) external {
        updatePool();
        UserInfo storage user = userInfo[msg.sender];
        if (user.amount > 0) {
            uint256 pending = (user.amount * accRewardPerShare) / ACC_PRECISION - user.rewardDebt;
            if (pending > 0) {
                _safeRewardTransfer(msg.sender, pending);
                emit Harvest(msg.sender, pending);
            }
        }
        user.rewardDebt = (user.amount * accRewardPerShare) / ACC_PRECISION;
        if (amount > 0) {
            require(lpToken.transferFrom(msg.sender, address(this), amount), "transfer");
            user.amount += amount;
            totalStaked += amount;
        }
        emit Deposit(msg.sender, amount);
    }

    function withdraw(uint256 amount) external {
        UserInfo storage user = userInfo[msg.sender];
        require(user.amount >= amount, "amount");
        updatePool();
        uint256 pending = (user.amount * accRewardPerShare) / ACC_PRECISION - user.rewardDebt;
        if (pending > 0) {
            _safeRewardTransfer(msg.sender, pending);
            emit Harvest(msg.sender, pending);
        }
        if (amount > 0) {
            user.amount -= amount;
            totalStaked -= amount;
            require(lpToken.transfer(msg.sender, amount), "transfer");
        }
        user.rewardDebt = (user.amount * accRewardPerShare) / ACC_PRECISION;
        emit Withdraw(msg.sender, amount);
    }

    function harvest() external {
        updatePool();
        UserInfo storage user = userInfo[msg.sender];
        uint256 accumulated = (user.amount * accRewardPerShare) / ACC_PRECISION;
        uint256 pending = accumulated - user.rewardDebt;
        user.rewardDebt = accumulated;
        if (pending > 0) {
            _safeRewardTransfer(msg.sender, pending);
            emit Harvest(msg.sender, pending);
        }
    }

    function emergencyWithdraw() external {
        UserInfo storage user = userInfo[msg.sender];
        uint256 amount = user.amount;
        user.amount = 0;
        user.rewardDebt = 0;
        totalStaked -= amount;
        require(lpToken.transfer(msg.sender, amount), "transfer");
        emit EmergencyWithdraw(msg.sender, amount);
    }

    function setRewardPerSecond(uint256 _rewardPerSecond) external {
        require(msg.sender == owner, "owner");
        updatePool();
        rewardPerSecond = _rewardPerSecond;
        emit RewardRateUpdated(_rewardPerSecond);
    }

    function _safeRewardTransfer(address to, uint256 amount) internal {
        uint256 balance = rewardToken.balanceOf(address(this));
        if (amount > balance) amount = balance;
        require(rewardToken.transfer(to, amount), "transfer");
    }
}

contract StreamPayC11 {
    struct Stream {
        address sender;
        address recipient;
        uint128 deposit;
        uint128 withdrawn;
        uint256 ratePerSecond;
        uint40 startTime;
        uint40 stopTime;
        bool cancelled;
    }

    TokenC11 public immutable token;
    uint256 public nextStreamId = 1;
    mapping(uint256 => Stream) public streams;

    event StreamCreated(uint256 indexed id, address indexed sender, address indexed recipient, uint256 deposit, uint256 rate, uint256 start, uint256 stop);
    event Withdrawn(uint256 indexed id, address indexed recipient, uint256 amount);
    event Cancelled(uint256 indexed id, uint256 senderRefund, uint256 recipientPaid);

    constructor(TokenC11 _token) {
        token = _token;
    }

    function createStream(address recipient, uint256 ratePerSecond, uint40 startTime, uint40 stopTime)
        external
        returns (uint256 id)
    {
        require(recipient != address(0) && recipient != address(this), "recipient");
        require(ratePerSecond > 0, "rate");
        require(startTime >= block.timestamp, "start");
        require(stopTime > startTime, "stop");

        uint256 duration = uint256(stopTime) - startTime;
        uint128 deposit = uint128(ratePerSecond * duration);

        id = nextStreamId++;
        streams[id] = Stream({
            sender: msg.sender,
            recipient: recipient,
            deposit: deposit,
            withdrawn: 0,
            ratePerSecond: ratePerSecond,
            startTime: startTime,
            stopTime: stopTime,
            cancelled: false
        });

        require(token.transferFrom(msg.sender, address(this), deposit), "transfer");
        emit StreamCreated(id, msg.sender, recipient, deposit, ratePerSecond, startTime, stopTime);
    }

    function streamed(uint256 id) public view returns (uint256) {
        Stream storage s = streams[id];
        if (block.timestamp <= s.startTime) return 0;
        uint256 end = block.timestamp < s.stopTime ? block.timestamp : s.stopTime;
        return (end - s.startTime) * s.ratePerSecond;
    }

    function withdrawable(uint256 id) public view returns (uint256) {
        Stream storage s = streams[id];
        if (s.cancelled) return 0;
        return streamed(id) - s.withdrawn;
    }

    function withdraw(uint256 id, uint128 amount) external {
        Stream storage s = streams[id];
        require(msg.sender == s.recipient, "recipient");
        require(amount > 0 && amount <= withdrawable(id), "amount");
        s.withdrawn += amount;
        require(token.transfer(s.recipient, amount), "transfer");
        emit Withdrawn(id, s.recipient, amount);
    }

    function cancel(uint256 id) external {
        Stream storage s = streams[id];
        require(msg.sender == s.sender || msg.sender == s.recipient, "party");
        require(!s.cancelled, "cancelled");
        uint256 owedToRecipient = streamed(id) - s.withdrawn;
        uint256 refund = uint256(s.deposit) - s.withdrawn - owedToRecipient;
        s.cancelled = true;
        s.withdrawn += uint128(owedToRecipient);
        if (owedToRecipient > 0) require(token.transfer(s.recipient, owedToRecipient), "transfer");
        if (refund > 0) require(token.transfer(s.sender, refund), "transfer");
        emit Cancelled(id, refund, owedToRecipient);
    }
}

contract AuctionHouseC12 {
    struct Auction {
        address seller;
        uint96 lotAmount;
        uint64 endTime;
        bool settled;
        address highestBidder;
        uint256 highestBid;
        uint256 reservePrice;
    }

    uint256 public constant MIN_INCREMENT_BPS = 500;
    uint256 public constant EXTENSION_WINDOW = 10 minutes;
    uint256 public constant MAX_DURATION = 30 days;

    TokenC12 public immutable lot;
    uint256 public nextAuctionId = 1;
    mapping(uint256 => Auction) public auctions;
    mapping(address => uint256) public pendingWithdrawals;

    uint256 private locked = 1;

    event AuctionCreated(uint256 indexed id, address indexed seller, uint256 amount, uint256 reserve, uint256 endTime);
    event BidPlaced(uint256 indexed id, address indexed bidder, uint256 amount, uint256 endTime);
    event AuctionSettled(uint256 indexed id, address winner, uint256 amount);
    event AuctionCancelled(uint256 indexed id);
    event Withdrawal(address indexed account, uint256 amount);

    constructor(TokenC12 _lot) {
        lot = _lot;
    }

    modifier nonReentrant() {
        require(locked == 1, "reentrant");
        locked = 2;
        _;
        locked = 1;
    }

    function createAuction(uint96 lotAmount, uint256 reservePrice, uint64 duration)
        external
        nonReentrant
        returns (uint256 id)
    {
        require(lotAmount > 0, "amount");
        require(duration > 0 && duration <= MAX_DURATION, "duration");
        uint256 beforeBal = lot.balanceOf(address(this));
        require(lot.transferFrom(msg.sender, address(this), lotAmount), "transfer");
        require(lot.balanceOf(address(this)) - beforeBal == lotAmount, "received");

        id = nextAuctionId++;
        auctions[id] = Auction({
            seller: msg.sender,
            lotAmount: lotAmount,
            endTime: uint64(block.timestamp) + duration,
            settled: false,
            highestBidder: address(0),
            highestBid: 0,
            reservePrice: reservePrice
        });
        emit AuctionCreated(id, msg.sender, lotAmount, reservePrice, block.timestamp + duration);
    }

    function minNextBid(uint256 id) public view returns (uint256) {
        Auction storage a = auctions[id];
        if (a.highestBid == 0) return a.reservePrice == 0 ? 1 : a.reservePrice;
        return a.highestBid + (a.highestBid * MIN_INCREMENT_BPS + 9_999) / 10_000;
    }

    function bid(uint256 id) external payable nonReentrant {
        Auction storage a = auctions[id];
        require(a.seller != address(0), "no auction");
        require(!a.settled, "settled");
        require(block.timestamp < a.endTime, "ended");
        require(msg.sender != a.seller, "seller");
        require(msg.value >= minNextBid(id), "bid too low");

        address previous = a.highestBidder;
        uint256 previousBid = a.highestBid;

        a.highestBidder = msg.sender;
        a.highestBid = msg.value;

        if (previous != address(0)) {
            pendingWithdrawals[previous] += previousBid;
        }

        if (a.endTime - block.timestamp < EXTENSION_WINDOW) {
            a.endTime = uint64(block.timestamp + EXTENSION_WINDOW);
        }
        emit BidPlaced(id, msg.sender, msg.value, a.endTime);
    }

    function settle(uint256 id) external nonReentrant {
        Auction storage a = auctions[id];
        require(a.seller != address(0), "no auction");
        require(!a.settled, "settled");
        require(block.timestamp >= a.endTime, "not ended");
        a.settled = true;

        if (a.highestBidder == address(0)) {
            require(lot.transfer(a.seller, a.lotAmount), "transfer");
            emit AuctionSettled(id, address(0), 0);
            return;
        }

        pendingWithdrawals[a.seller] += a.highestBid;
        require(lot.transfer(a.highestBidder, a.lotAmount), "transfer");
        emit AuctionSettled(id, a.highestBidder, a.highestBid);
    }

    function cancel(uint256 id) external nonReentrant {
        Auction storage a = auctions[id];
        require(msg.sender == a.seller, "seller");
        require(!a.settled, "settled");
        require(a.highestBidder == address(0), "has bids");
        a.settled = true;
        require(lot.transfer(a.seller, a.lotAmount), "transfer");
        emit AuctionCancelled(id);
    }

    function withdraw() external nonReentrant {
        uint256 amount = pendingWithdrawals[msg.sender];
        require(amount > 0, "nothing");
        pendingWithdrawals[msg.sender] = 0;
        (bool ok, ) = msg.sender.call{value: amount}("");
        require(ok, "send failed");
        emit Withdrawal(msg.sender, amount);
    }
}
